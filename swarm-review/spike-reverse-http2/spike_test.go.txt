package spike

import (
	"bufio"
	"context"
	"fmt"
	"io"
	"net"
	"net/http"
	"net/http/httptest"
	"strings"
	"sync"
	"testing"
	"time"

	"golang.org/x/net/http2"
)

// TestReverseHTTP2Tunnel proves a node behind NAT can dial OUT to a relay and
// still serve its whole HTTP surface (unary + SSE) back through that one conn.
func TestReverseHTTP2Tunnel(t *testing.T) {
	// --- the node's real handler (stands in for coddy http) ---
	nodeMux := http.NewServeMux()
	nodeMux.HandleFunc("/coddy/sessions", func(w http.ResponseWriter, r *http.Request) {
		fmt.Fprintf(w, `{"auth":%q,"method":%q}`, r.Header.Get("Authorization"), r.Method)
	})
	nodeMux.HandleFunc("/v1/responses", func(w http.ResponseWriter, r *http.Request) {
		body, _ := io.ReadAll(r.Body)
		w.Header().Set("Content-Type", "text/event-stream")
		fl, ok := w.(http.Flusher)
		if !ok {
			t.Error("no flusher on node side")
			return
		}
		for i := 0; i < 3; i++ {
			fmt.Fprintf(w, "data: chunk-%d-%s\n\n", i, strings.TrimSpace(string(body)))
			fl.Flush()
			time.Sleep(20 * time.Millisecond)
		}
		fmt.Fprint(w, "data: [DONE]\n\n")
		fl.Flush()
	})

	// --- the relay: accepts the dial-out and keeps the RoundTripper ---
	var mu sync.Mutex
	var rt http.RoundTripper
	ready := make(chan struct{})

	relay := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if r.URL.Path == "/swarm/tunnel" {
			hj, ok := w.(http.Hijacker)
			if !ok {
				http.Error(w, "no hijack", 500)
				return
			}
			conn, brw, err := hj.Hijack()
			if err != nil {
				t.Error(err)
				return
			}
			// tell the node the tunnel is accepted
			_, _ = brw.WriteString("HTTP/1.1 200 OK\r\nContent-Length: 0\r\n\r\n")
			_ = brw.Flush()
			tr := &http2.Transport{AllowHTTP: true, ReadIdleTimeout: 30 * time.Second}
			cc, err := tr.NewClientConn(conn)
			if err != nil {
				t.Error(err)
				return
			}
			mu.Lock()
			rt = cc
			mu.Unlock()
			close(ready)
			return
		}
		http.NotFound(w, r)
	}))
	defer relay.Close()

	// --- the node dials out ---
	go func() {
		addr := strings.TrimPrefix(relay.URL, "http://")
		conn, err := net.Dial("tcp", addr)
		if err != nil {
			t.Error(err)
			return
		}
		req, _ := http.NewRequest("POST", relay.URL+"/swarm/tunnel", nil)
		if err := req.Write(conn); err != nil {
			t.Error(err)
			return
		}
		br := bufio.NewReader(conn)
		resp, err := http.ReadResponse(br, req)
		if err != nil {
			t.Error(err)
			return
		}
		if resp.StatusCode != 200 {
			t.Errorf("tunnel handshake: %s", resp.Status)
			return
		}
		// serve our own HTTP surface over the hijacked conn
		(&http2.Server{}).ServeConn(conn, &http2.ServeConnOpts{Handler: nodeMux})
	}()

	select {
	case <-ready:
	case <-time.After(5 * time.Second):
		t.Fatal("tunnel never established")
	}

	mu.Lock()
	tunnel := rt
	mu.Unlock()

	// --- 1. unary request through the tunnel ---
	req, _ := http.NewRequest("GET", "http://node.invalid/coddy/sessions", nil)
	req.Header.Set("Authorization", "Bearer node-token")
	resp, err := tunnel.RoundTrip(req)
	if err != nil {
		t.Fatalf("unary roundtrip: %v", err)
	}
	body, _ := io.ReadAll(resp.Body)
	_ = resp.Body.Close()
	if !strings.Contains(string(body), "Bearer node-token") {
		t.Fatalf("headers did not survive: %s", body)
	}
	t.Logf("unary ok: %s", body)

	// --- 2. SSE stream through the tunnel, chunk by chunk, no buffering ---
	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()
	sreq, _ := http.NewRequestWithContext(ctx, "POST", "http://node.invalid/v1/responses", strings.NewReader("hello"))
	sresp, err := tunnel.RoundTrip(sreq)
	if err != nil {
		t.Fatalf("sse roundtrip: %v", err)
	}
	defer func() { _ = sresp.Body.Close() }()
	sc := bufio.NewScanner(sresp.Body)
	var chunks []string
	start := time.Now()
	var firstAt time.Duration
	for sc.Scan() {
		line := strings.TrimSpace(sc.Text())
		if line == "" {
			continue
		}
		if len(chunks) == 0 {
			firstAt = time.Since(start)
		}
		chunks = append(chunks, line)
		if strings.Contains(line, "[DONE]") {
			break
		}
	}
	if len(chunks) != 4 {
		t.Fatalf("want 4 sse lines, got %d: %v", len(chunks), chunks)
	}
	if firstAt > 60*time.Millisecond {
		t.Fatalf("first chunk arrived after %v - stream was buffered", firstAt)
	}
	t.Logf("sse ok, first chunk in %v: %v", firstAt, chunks)

	// --- 3. concurrent multiplexing over the single conn ---
	var wg sync.WaitGroup
	errs := make(chan error, 8)
	for i := 0; i < 8; i++ {
		wg.Add(1)
		go func(n int) {
			defer wg.Done()
			r, _ := http.NewRequest("GET", "http://node.invalid/coddy/sessions", nil)
			r.Header.Set("Authorization", fmt.Sprintf("Bearer t%d", n))
			rs, err := tunnel.RoundTrip(r)
			if err != nil {
				errs <- err
				return
			}
			b, _ := io.ReadAll(rs.Body)
			_ = rs.Body.Close()
			if !strings.Contains(string(b), fmt.Sprintf("Bearer t%d", n)) {
				errs <- fmt.Errorf("stream %d crossed: %s", n, b)
			}
		}(i)
	}
	wg.Wait()
	close(errs)
	for e := range errs {
		t.Fatalf("concurrent: %v", e)
	}
	t.Log("8 concurrent multiplexed requests ok")
}
